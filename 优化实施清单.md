### 1.1 状态持久化 🔥 [2天]

**问题**：刷新页面导致状态丢失，用户体验极差。

**方案**：集成 Zustand + localStorage

```bash
npm install zustand
```

```typescript
// src/lib/store-v2.ts
import { create } from 'zustand';
import { persist, createJSONStorage } from 'zustand/middleware';

interface AppState {
  currentCustomer: Customer | null;
  currentKYC: CustomerKYC | null;
  currentPlan: AllocationPlan | null;
  setCustomer: (customer: Customer) => void;
  setKYC: (kyc: CustomerKYC) => void;
  setPlan: (plan: AllocationPlan) => void;
}

export const useAppStore = create<AppState>()(
  persist(
    (set) => ({
      currentCustomer: null,
      currentKYC: null,
      currentPlan: null,
      setCustomer: (customer) => set({ currentCustomer: customer }),
      setKYC: (kyc) => set({ currentKYC: kyc }),
      setPlan: (plan) => set({ currentPlan: plan }),
    }),
    {
      name: 'icbc-wmp-storage',
      storage: createJSONStorage(() => localStorage),
    }
  )
);
```

**迁移步骤**：
1. 安装依赖
2. 创建新store文件
3. 逐页替换 `useAppState()` 为 `useAppStore()`
4. 删除旧的 `src/lib/store.tsx`

**验证**：刷新页面后KYC信息和方案数据应保留。

```typescript
// src/app/plan/[planId]/compare/page.tsx
'use client';

import { useState } from 'react';
import { useAppStore } from '@/lib/store-v2';
import { dualTreeClassify } from '@/lib/algorithm/dual-tree';
import { generatePyramidPlan } from '@/lib/algorithm/pyramid';
import { computeCashFlow } from '@/lib/algorithm/cashflow';

export default function PlanComparePage({ params }) {
  const { currentKYC } = useAppStore();
  const [selectedRisks, setSelectedRisks] = useState(['R2', 'R3', 'R4']);
  
  if (!currentKYC) return <div>请先完成KYC评估</div>;
  
  // 生成3种风险偏好方案
  const plans = selectedRisks.map(riskLevel => {
    const modifiedKYC = { ...currentKYC, riskLevel };
    const tierResult = dualTreeClassify(modifiedKYC);
    const { config, items } = generatePyramidPlan(modifiedKYC, tierResult.finalTier);
    const cashFlowResult = computeCashFlow(modifiedKYC, config, items);
    
    return {
      riskLevel,
      label: riskLevel === 'R2' ? '保守' : riskLevel === 'R3' ? '平衡' : '激进',

      items,

  });
  
  return (
    <div className="max-w-7xl mx-auto px-8 py-10">
      <h1 className="section-title-lg mb-8">方案对比</h1>
      
      {/* 对比表格 */}
      <div className="grid grid-cols-3 gap-4">
        {plans.map((plan, idx) => (
          <div key={idx} className={`card card-padded ${idx === 1 ? 'border-2 border-[var(--brand-red)]' : ''}`}>
            <div className="text-center mb-4">
              <div className={`tag ${idx === 0 ? 'tag-positive' : idx === 1 ? 'tag-red' : 'tag-warning'} mb-2`}>
                {plan.label}方案
              </div>
              <div className="text-sm text-[var(--ink-2)]">风险等级 {plan.riskLevel}</div>
            </div>
            
            {/* 金字塔配比 */}

                <div className="flex justify-between">
                  <span>现金层</span>
                  <span className="font-semibold">{(plan.config.cashRatio * 100).toFixed(1)}%</span>
                </div>
                <div className="flex justify-between">
                  <span>稳健层</span>
                  <span className="font-semibold">{(plan.config.stableRatio * 100).toFixed(1)}%</span>
                </div>
                <div className="flex justify-between">
                  <span>成长层</span>
                  <span className="font-semibold">{(plan.config.growthRatio * 100).toFixed(1)}%</span>
                </div>
                <div className="flex justify-between">
                  <span>卫星层</span>
                  <span className="font-semibold">{(plan.config.satelliteRatio * 100).toFixed(1)}%</span>

            {/* 预期收益 */}

              <div className="text-xs text-[var(--ink-3)] mb-2">退休预测</div>
              <div className="text-2xl font-bold text-[var(--brand-red)] mb-1">
                ¥{(plan.cashFlowResult.assetsAtRetirement / 10000).toFixed(0)}万
              </div>
```

```typescript
// src/app/tools/dca-calculator/page.tsx
'use client';

import { useState, useMemo } from 'react';
import dynamic from 'next/dynamic';

const DCAChart = dynamic(() => import('@/components/charts/DCAChart'), { ssr: false });

export default function DCACalculatorPage() {
  const [monthly, setMonthly] = useState(3000);
  const [years, setYears] = useState(30);
  const [rate, setRate] = useState(8);
  
  const result = useMemo(() => {
    const n = years * 12;
    const r = rate / 100 / 12;
    const fv = r === 0 ? monthly * n : monthly * ((Math.pow(1 + r, n) - 1) / r);
    
    const yearlyData = [];
    for (let y = 1; y <= years; y++) {
      const months = y * 12;
      const value = r === 0 ? monthly * months : monthly * ((Math.pow(1 + r, months) - 1) / r);
      yearlyData.push({ year: y, value, contribution: monthly * months });
    }
    
    return {
      totalContribution: monthly * n,
      endingBalance: fv,
      earnings: fv - monthly * n,
      yearlyData,
    };
  }, [monthly, years, rate]);
  
  return (
    <div className="max-w-4xl mx-auto px-8 py-10">
      <h1 className="section-title-lg mb-2">定投计算器</h1>
      <p className="text-sm text-[var(--ink-2)] mb-8">
        模拟长期定投的复利效果，理解时间的价值
      </p>
      
      <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
        {/* 左侧：参数输入 */}
        <div className="card card-padded">
          <h3 className="section-title mb-4">投资参数</h3>
          
          <div className="space-y-4">
            <div>
              <label className="label">每月定投金额（元）</label>
              <input
                type="range"
                min="500"
                max="20000"
                step="500"
                value={monthly}
                onChange={e => setMonthly(Number(e.target.value))}
                className="w-full"
              />
              <div className="text-right text-2xl font-bold text-[var(--brand-red)] mt-1">
                ¥{monthly.toLocaleString()}
              </div>
            </div>
            
            <div>
              <label className="label">投资年限（年）</label>
              <input
                type="range"
                min="5"
                max="40"
                step="1"
                value={years}
                onChange={e => setYears(Number(e.target.value))}
                className="w-full"
              />
              <div className="text-right text-2xl font-bold text-[var(--brand-red)] mt-1">
                {years}年

                type="range"
                min="3"

                onChange={e => setRate(Number(e.target.value))}
                className="w-full"
              />
              <div className="text-right text-2xl font-bold text-[var(--brand-red)] mt-1">
                {rate.toFixed(1)}%
              </div>
            </div>
          </div>
        </div>
        
        {/* 右侧：结果展示 */}
        <div className="card card-padded bg-gradient-to-br from-[var(--brand-red)] to-[var(--brand-red-dark)] text-white">
          <h3 className="text-lg font-semibold mb-4">投资结果</h3>
          
          <div className="space-y-4">
            <div>
              <div className="text-sm opacity-80">累计投入</div>

                ¥{result.totalContribution.toLocaleString()}
              </div>
            </div>
            
            <div>
              <div className="text-sm opacity-80">投资收益</div>
              <div className="text-2xl font-bold text-[var(--brand-gold-light)]">
```

```typescript
import DashScope from '@alicloud/dashscope';

const client = new DashScope({
  apiKey: process.env.DASHSCOPE_API_KEY!,
});

export async function chatWithQwen(messages: { role: string; content: string }[]) {
  const response = await client.generation.call({
    model: 'qwen-plus',
    input: { messages },
    parameters: {
      temperature: 0.7,
      max_tokens: 1500,
      result_format: 'message',
    },
  });
  
  return response.output.choices[0].message.content;
}
```

```typescript
// src/app/api/chat/route.ts
import { NextRequest, NextResponse } from 'next/server';
import { chatWithQwen } from '@/lib/llm/qwen';

const SYSTEM_PROMPT = `你是工商银行智能投顾"智盈"的AI助手...（见llm-advisor.ts）`;

export async function POST(req: NextRequest) {
  const { messages, kyc, plan } = await req.json();
  
  // 构造增强context
  const context = `
客户档案：年龄${kyc.age}岁，风险承受等级${kyc.riskLevel}，可投资资产${kyc.investableAssets}元
当前方案：现金${plan.config.cashRatio}%、稳健${plan.config.stableRatio}%、成长${plan.config.growthRatio}%、卫星${plan.config.satelliteRatio}%
`;
  
  const fullMessages = [
    { role: 'system', content: SYSTEM_PROMPT + '\n\n' + context },
    ...messages,
  ];
  
  try {
    const answer = await chatWithQwen(fullMessages);
    return NextResponse.json({ answer });
  } catch (error) {
    return NextResponse.json({ error: error.message }, { status: 500 });
  }
}
```
